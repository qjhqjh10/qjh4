### 2·2 `A1961`：比较函数（`BattleDriver.cs:6152`）

```csharp
static int CompareChoiceOptions(CardDef a, CardDef b)
{
    int c = a.Cost.CompareTo(b.Cost);                              // ① manaCost @0x78
    if (c != 0) return c;
    string an = CardText.Name(a), bn = CardText.Name(b);           // ② 本地化卡名（= 面板显示的那个）
    c = string.Compare((an ?? "").ToLower(), (bn ?? "").ToLower(),
                       System.StringComparison.CurrentCulture);    //    原版 `.ToLower().CompareTo()`
    if (c != 0) return c;
    return string.Compare(a.Id ?? "", b.Id ?? "",
                          System.StringComparison.CurrentCulture); // ③ uniqueId @0x38
}
```
