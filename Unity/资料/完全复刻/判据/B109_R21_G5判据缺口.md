### 复现 N54 尺子的脚本（纯 Python，可直接跑）

```bash
python -I -X utf8 -c "
import json
d=json.load(open('d:/4/Unity/MyGame/Assets/RuleEngine/Resources/cards_truth.json',encoding='utf-8'))
TR=[0,80,81,120,430,440,145,130,230,45,46,220,300,301,550,560,580,590,655,670,690,680,710,715,720]
TV=[10,120,1010,260,760,770,1000,280,450,840,845,470,920,1225,960,1040,1110,1130,1185,1200,1230,1210,1245,1250,1255]
FAC={10:1275,30:1185,50:1220,60:1235,70:1265}
for ct in (None,20):
    n=m=0
    for c in d['cards']:
        if ct and c.get('cardType')!=ct: continue
        army=c.get('army'); tr=set(a['trigger'] for a in (c.get('abilities') or []) if a.get('trigger') is not None)
        der=[tv for t,tv in zip(TR,TV) if army not in (20,40) and t in tr]
        if c.get('hasActiveAbility')==1: der.append(FAC.get(army,400))
        orig=set(t['_traitValue'] for t in (c.get('cardTraits') or []))
        ex=[x for x in der if x not in orig]
        if ex: n+=1; m+=len(ex)
    print('cardType filter',ct,'->',n,m)
"
```
