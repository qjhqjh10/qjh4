        {

            foreach (var r in c.abilityEffectTexts)
            {
                if (r == null) continue;

                                 + $"(ab={r.ab}, lg={r.lg}, isHand={r.isHand})** —— 后一条已丢弃。重跑 工具/gen_cards_engine.py。");
                    continue;
