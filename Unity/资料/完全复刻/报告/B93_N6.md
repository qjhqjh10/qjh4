### B. `UICornersGradient` 的四角双线性 + **乘**顶点原色

**插哪节**：`CardPresentation/Editor/ShellScene.cs`（共用 UI 件那一族）或 `Editor/BattleScene.cs` 任一处**末尾**。

```csharp
// ---- B93_N6 · 账 未分类_其他:667 —— `UICornersGradient.ModifyMesh` 的四角双线性 ----
{
    var go = new GameObject("B93N6_CornerGradientProbe", typeof(RectTransform), typeof(UnityEngine.UI.Image));
    var rt = (RectTransform)go.transform;
    rt.sizeDelta = new Vector2(100f, 100f);                       // rect = (-50,-50)..(50,50)
    var cg = go.AddComponent<CardPresentation.UICornersGradient>();
    cg.m_bottomLeftColor  = Color.red;                            // 左下 红
    cg.m_bottomRightColor = Color.green;                          // 右下 绿
    cg.m_topLeftColor     = Color.blue;                           // 左上 蓝
    cg.m_topRightColor    = Color.white;                          // 右上 白
    Rect r = rt.rect;                                             // ⚠️ 非拉伸锚 ⇒ rect.rect 就是 sizeDelta
    Color32 white = new Color32(255, 255, 255, 255);
    Color32 bl = cg.ColorAt(r.xMin, r.yMin, white, r);            // 左下角
    Color32 br = cg.ColorAt(r.xMax, r.yMin, white, r);            // 右下角
    Color32 tr = cg.ColorAt(r.xMax, r.yMax, white, r);            // 右上角
    Color32 mid = cg.ColorAt(0f, 0f, white, r);                   // 中心
    Color32 half = cg.ColorAt(r.xMin, r.yMin, new Color32(255, 255, 255, 128), r);   // α 只有一半
    CheckTrue(bl.r > 250 && bl.g < 5 && bl.b < 5,   "[B93_N6] 左下角取 `m_bottomLeftColor`（红）");
    CheckTrue(br.g > 250 && br.r < 5 && br.b < 5,   "[B93_N6] 右下角取 `m_bottomRightColor`（绿）");
    CheckTrue(tr.b > 250 && tr.r > 250 && tr.g > 250, "[B93_N6] 右上角取 `m_topRightColor`（白）");
    CheckTrue(Mathf.Abs(mid.r - 127) <= 1 && Mathf.Abs(mid.g - 127) <= 1 && Mathf.Abs(mid.b - 127) <= 1,
              "[B93_N6] 中心 = 四角双线性中点（三通道各 ≈127）");
    CheckTrue(half.a >= 127 && half.a <= 128,
              "[B93_N6] **乘**顶点原色（α 128 不会被重置成 255）—— 这是「乘不是替换」的判别式");
    UnityEngine.Object.DestroyImmediate(go);
}
```
