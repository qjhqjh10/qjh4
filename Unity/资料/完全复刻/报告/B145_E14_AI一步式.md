```csharp
// 新增（类尾）
public static bool PlayTurnStep(BattleContext ctx, AiDifficulty diff, IList<AiAction> rejected)
{
    if (ctx == null || ctx.IsOver) return false;
    AiAction a;
    if (!NextAction(ctx, diff, rejected, out a)) return false;   // 最佳是 endTurn / 没有动作可做
    if (ExecuteAction(ctx, a)) { if (rejected != null) rejected.Clear(); return true; }
    if (rejected == null) return false;                          // 没有排除表 ⇒ 收手（否则死循环）
    rejected.Add(a);
    return rejected.Count < MaxRejectedActions;
}
public static bool PlayTurnStep(BattleContext ctx, AiDifficulty diff)          // = 上者 rejected:null
public const int PlayTurnGuard = 64;                                          // 兼容壳的步数上限
```
